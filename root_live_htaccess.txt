# Configuração para Apache com suporte a SPA (Single Page Application)

AddDefaultCharset UTF-8

<IfModule mod_mime.c>
  AddType 'text/html; charset=UTF-8' .html
  AddType 'text/css; charset=UTF-8' .css
  AddType 'application/javascript; charset=UTF-8' .js
</IfModule>

<IfModule mod_rewrite.c>

  RewriteEngine On

  RewriteBase /

  

  # Se for um arquivo ou diretório real, não reescreva

  RewriteCond %{REQUEST_FILENAME} !-f

  RewriteCond %{REQUEST_FILENAME} !-d

  

  # Reescreva para index.html

  RewriteRule ^ index.html [QSA,L]

</IfModule>



# Headers para cache e segurança

<IfModule mod_headers.c>

  # Permitir iframe de outros domínios

  Header always set X-Frame-Options "ALLOWALL"

  

  # Permitir acesso CORS (se necessário)

  Header set Access-Control-Allow-Origin "*"

  

  # Cache control

  <FilesMatch "\.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf|eot)$">

    Header set Cache-Control "max-age=31536000, public"

  </FilesMatch>

  

  <FilesMatch "\.(html)$">

    Header set Cache-Control "max-age=3600, public"

  </FilesMatch>

</IfModule>



# Compressão Gzip

<IfModule mod_deflate.c>

  AddOutputFilterByType DEFLATE text/html text/plain text/css text/javascript application/javascript

</IfModule>
